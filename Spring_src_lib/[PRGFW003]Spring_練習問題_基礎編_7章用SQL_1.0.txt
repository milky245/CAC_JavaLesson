-- 産地テーブルの作成 
CREATE TABLE area ( 
area_id NUMBER(2) PRIMARY KEY, 
area_name VARCHAR2(10 CHAR) NOT NULL,
detail VARCHAR2(30 CHAR)
); 


-- 果物の産地テーブルの作成
CREATE TABLE fruits_area (
  fruit_id NUMBER(5) PRIMARY KEY, -- 識別番号
  fruit_name VARCHAR2(20 CHAR) NOT NULL, -- 果物の名前
  area_id NUMBER(2) REFERENCES area(area_id) NOT NULL-- 外部参照
 );
